{{- /* Previous / next links, following the reading order of the book. */ -}} {{- $book := .FirstSection -}} {{- $pages := (partialCached "book.html" $book $book.Path).pages -}} {{- $current := . -}} {{- $prev := false -}} {{- $next := false -}} {{- range $i, $page := $pages -}} {{- if eq $page $current -}} {{- if gt $i 0 }}{{ $prev = index $pages (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $pages) }}{{ $next = index $pages (add $i 1) }}{{ end -}} {{- end -}} {{- end -}} {{- if or $prev $next }}
{{- with $prev }} ← {{ .LinkTitle }} {{- end }} {{- with $next }} {{ .LinkTitle }} → {{- end }}
{{- end }}
